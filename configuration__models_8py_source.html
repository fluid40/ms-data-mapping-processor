54def load_configuration_file(config_file: Path) -> ServiceConfiguration | None:
55 """Load the runtime configuration from a JSON file.
56
57 :param config_file_path: Path to the configuration file
58 :return: ServiceConfiguration object if successful, None otherwise
59 """
60 if config_file is None:
61 _logger.error("No configuration file provided.")
62 return None
63
64 config_file = config_file.resolve()
65 _logger.info(f"Load configuration file '{config_file}'.")
66
67 if not config_file.exists() or not config_file.is_file():
68 _logger.error(f"Configuration file '{config_file}' not found or inaccessible. ")
69 return None
70
71 config_string = config_file.read_text(encoding="utf-8")
72 _logger.debug(f"Configuration file '{config_file}' found.")
73
74 try:
75 return ServiceConfiguration.model_validate_json(config_string)
76 except ValidationError as ve:
77 _logger.error(f"Invalid BaSyx server connection file: {ve}")
78 return None

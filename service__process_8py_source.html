21 :param mapping_configurations: The mapping configurations.
22 :param asset_connector: The asset connector client.
23 :param influx_client: The InfluxDB client.

27 asset_values: list[PayloadValue] = []

30 for relation in configuration.source_sink_relations:

95def _get_submodel_from_cache(
96 submodel_id: str, server_handler: ServerHandler, submodel_cache: dict[str, model.Submodel], descriptor_mapping: list[DescriptorMapping]
97) -> model.Submodel | None:
98 """Get a submodel from the cache.
99
100 :param submodel_id: The ID of the submodel.
101 :param server_handler: The server handler.
102 :param submodel_cache: The submodel cache.
103 :return: The submodel if found, None otherwise.
104 """
105 # Check if submodel is already in cache
106 if submodel_id in submodel_cache:
107 # Return cached submodel

110 # Fetch submodel from AAS server

117

119 submodel_cache[submodel_id] = submodel

121 return submodel

124def _rewrite_registry_descriptor(submodel_id: str, server_handler: ServerHandler, descriptor_mapping: list[DescriptorMapping]):
125 # get current descriptor from registry as 'master'
126 master_descriptor = server_handler.sm_registry_client.submodel_registry.get_submodel_descriptor_by_id(submodel_id)
127
128 if master_descriptor is None:
129 _logger.warning(f"Submodel descriptor with ID '{submodel_id}' not found in AAS registry. Cannot rewrite descriptor.")
130 return
131
132 # parse 'master' descriptor to get endpoint href and protocol information
133 master_href = descriptor_json_helper.get_endpoint_href_by_index(master_descriptor)
134 master_href_data = descriptor_json_helper.parse_endpoint_href(master_href)
135
136 # copy 'master' descriptor to 'slave' descriptor which will be rewritten and updated in registry
137 slave_descriptor = copy.deepcopy(master_descriptor)
138
139 # get the protocol information from the 'slave'
140 slave_protocol_info = descriptor_json_helper.get_endpoint_protocol_information_by_index(slave_descriptor)
141 # create new href with external URL and port and parsed tag and identifier
142 slave_href = f"{_get_base_url()}/{master_href_data.tag}/{master_href_data.identifier}"
143 slave_protocol_info["href"] = slave_href
144
145 # update descriptor with new href on registry server
146 _logger.debug(f"Update descriptor for submodel '{submodel_id}' with new href '{slave_href}' on registry server")
147 server_handler.sm_registry_client.submodel_registry.put_submodel_descriptor_by_id(submodel_id, slave_descriptor)
148
149 # save 'master' and 'slave' descriptor as a mapping for later use when submodels are fetched and descriptors need to be rewritten
150 mapping = DescriptorMapping(master_descriptor, slave_descriptor, submodel_id)
151 descriptor_mapping.append(mapping)

164def _convert_value(value: str, value_type: DataTypeDefXsd) -> any:
165 """Convert a string value to an expected value type.
166
167 :param value: string value
168 :param expected_type: expected value type
169 :raises ValueError: if the value cannot be converted to the expected type
170 :return: converted value
171 """
172 return_value: any = value
173
174 match value_type:
175 case model.datatypes.Boolean:
176 v = value.strip().lower()
177 if v == "true":
178 return_value = True
179 elif v == "false":
180 return_value = False
181 else:
182 raise ValueError(f"Cannot convert '{value}' to boolean.")
183 case model.datatypes.Integer | model.datatypes.Short:
184 return_value = int(value)
185 case model.datatypes.Double | model.datatypes.Float:
186 return_value = float(value)
187 case model.datatypes.DateTime:
188 return_value = datetime.datetime.fromisoformat(value)
189
190 return return_value

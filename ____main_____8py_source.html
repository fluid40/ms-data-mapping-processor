38) -> None:
39 """
40 Background worker that periodically polls asset values.
41
42 The worker cooperatively shuts down as soon as stop_event is set.
43 """
44 _logger.info("Worker started")
45
46 try:
47 while not stop_event.is_set():
48 try:
49 get_asset_values(states)
50 except Exception:
51 _logger.exception("Error while processing asset values")
52
53 # Wait for either shutdown or next polling interval
54 with suppress(TimeoutError):
55 await asyncio.wait_for(stop_event.wait(), timeout=interval)

57 finally:
58 _logger.info("Worker stopped")

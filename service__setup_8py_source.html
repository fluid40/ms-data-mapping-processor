23def setup_service(configuration: ServiceConfiguration) -> ServiceStates:
24 """Setup the service states with the given configuration.
25
26 :param configuration: The service configuration to use.
27 :return: An instance of ServiceStates with the provided configuration.
28 """
29 if configuration is None:
30 _logger.error("No configuration provided for service setup.")
31 raise HTTPException(status_code=StatusCode.CONFIGURATION_ERROR.value, detail="No configuration provided for service setup.")
32
33 set_external_url_and_port(configuration.external_url, str(configuration.external_port))
34
35 server_configurations = ServerConfigurationsHandler()
36
37 server_handler = ServerHandler()
38 server_handler.connect_to_server(server_configurations)
39
40 # get the AAS from server
41 shell: model.AssetAdministrationShell = get_shell(server_handler, configuration.aas_id)
42
43 # get the AIMC submodel from server
44 aimc_submodel: model.Submodel = _get_aimc_submodel(server_handler, shell)
45
46 # parse the AIMC submodel to get mapping configurations
47 mapping_configurations: MappingConfigurations = _get_mapping_configurations(aimc_submodel)
48
49 # get the AID submodels from the AAS
50 aid_submodels: list[model.Submodel] = _get_aid_submodels(server_handler, mapping_configurations.aid_submodel_ids)
51
52 # connect to Asset Connector
53 asset_connector_client = _connect_to_asset_connector(server_configurations.asset_connector_configuration)
54
55 # configure Asset Connector with AID submodels
56 _configure_asset_connector(asset_connector_client, aid_submodels)
57
58 # create process data
59 process_data = ProcessData(id=shell.id, id_short=shell.id_short, display_name=str(shell.display_name))
60
61 # create service states
62 service_states = ServiceStates(process_data)
63 service_states.service_configuration = configuration

65 service_states.asset_connector_client = asset_connector_client

71def _get_aimc_submodel(server_handler: ServerHandler, shell: model.AssetAdministrationShell) -> model.Submodel:
72 """Get the Asset Interface Mapping Configuration (AIMC) submodel from the AAS.
73
74 :param aas_server_wrapper: The SDK wrapper for the AAS server
75 :param shell: The AAS to get the submodel from
76 :raises HTTPException: If the AIMC submodel could not be found
77 :return: The AIMC submodel
78 """
79 _logger.info("Get AIMC submodel from Shell.")
80 submodel_ids = aas_parser.get_submodel_ids(shell)
81
82 for submodel_id in submodel_ids:
83 submodel = get_submodel(server_handler, submodel_id)
84
85 semantic_id_value = submodel_parser.get_semantic_id_value(submodel)
86
87 if semantic_id_value and "/idta/AssetInterfacesMappingConfiguration" in semantic_id_value:
88 _logger.info(f"AIMC submodel with ID '{submodel_id}' found on server.")
89 return submodel
90
91 _logger.error("No Submodel with semantic ID '/idta/AssetInterfacesMappingConfiguration' not found on server.")
92 raise HTTPException(
93 status_code=StatusCode.AIMC_NOT_FOUND.value,
94 detail="No Submodel with semantic ID '/idta/AssetInterfacesMappingConfiguration' not found on server.",
95 )

164def _configure_asset_connector(asset_connector_client: AssetConnectorClient, aid_submodels: list[model.Submodel]):
165 """Configure the Asset Connector with the provided AID submodels.
166
167 :param client: The Asset Connector client
168 :param aid_submodels: The list of AID submodels
169 """
170 _logger.info(f"Add {len(aid_submodels)} AID submodel configurations to Asset Connector.")
171 for aid_submodel in aid_submodels:
172 data_string = json.dumps(aid_submodel, cls=basyx.aas.adapter.json.AASToJsonEncoder)
173 mqtt_config = {"Aid": json.loads(data_string)}
174
175 _logger.info(f"Add AID submodel '{aid_submodel.id_short}' configuration to Asset Connector.")
176
177 result = asset_connector_client.add_config(mqtt_config)
178
179 if result is None or result.get_success() is False:
180 _logger.error(f"Could not add AID submodel '{aid_submodel.id_short}' configuration to Asset Connector.")
181 raise HTTPException(
182 status_code=StatusCode.AID_CONFIGURATION_ERROR.value,
183 detail=f"Could not add AID submodel '{aid_submodel.id_short}' configuration to Asset Connector.",
